CREATE PROCEDURE SP_GetUsersList_Nature
AS
BEGIN
	SET NOCOUNT ON;

	SELECT UserId, FullName, Email, ISNULL(IsBlocked, 0) AS IsBlocked
	FROM Users_Nature;
END
GO
