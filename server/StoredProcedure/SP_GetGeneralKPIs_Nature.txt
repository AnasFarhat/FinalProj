CREATE PROCEDURE SP_GetGeneralKPIs_Nature
AS
BEGIN
	SET NOCOUNT ON;

	SELECT 
		ISNULL((SELECT COUNT(*) FROM Users_Nature), 0) AS TotalUsers,
		ISNULL((SELECT CAST(AVG(CAST((GuideRating + TrackRating) / 2.0 AS FLOAT)) AS DECIMAL(10,2)) 
				FROM Feedbacks_Nature), 0) AS AvgSatisfaction,
		ISNULL((SELECT COUNT(DISTINCT UserId) FROM UserTrips_Nature), 0) AS ActiveHikers;
END
GO
