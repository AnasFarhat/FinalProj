CREATE PROCEDURE SP_ReportContent_Nature
	@PostId INT = NULL,
	@CommentId INT = NULL,
	@UserId INT,
	@ReasonCategory NVARCHAR(100),
	@CustomReason NVARCHAR(MAX) = NULL
AS
BEGIN
	SET NOCOUNT ON;

	INSERT INTO Reports_Nature (PostId, CommentId, UserId, ReasonCategory, CustomReason, CreatedAt)
	VALUES (@PostId, @CommentId, @UserId, @ReasonCategory, @CustomReason, GETDATE());


END
GO
