CREATE PROCEDURE SP_GetFeedbackHistoryByUserId_Nature
	@UserId INT
AS
BEGIN
	SET NOCOUNT ON;

	SELECT f.FeedbackId, 
	       f.TripId, 
	       f.GuideRating, 
	       f.TrackRating, 
	       f.[FreeText], 
	       f.CreatedAt, 
	       f.VoiceFilePath, 
	       t.Title AS TripTitle
	FROM Feedbacks_Nature f
	JOIN Trips_Nature t ON f.TripId = t.TripId
	WHERE f.UserId = @UserId
	ORDER BY f.CreatedAt DESC;
END
GO
