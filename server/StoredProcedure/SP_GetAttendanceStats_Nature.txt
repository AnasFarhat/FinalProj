CREATE PROCEDURE SP_GetAttendanceStats_Nature
AS
BEGIN
	SET NOCOUNT ON;

	SELECT t.Title, 
		   COUNT(ut.UserId) AS RegisteredCount,
		   SUM(CASE WHEN ut.Status = N'הסתיים' THEN 1 ELSE 0 END) AS ArrivedCount
	FROM Trips_Nature t
	LEFT JOIN UserTrips_Nature ut ON t.TripId = ut.TripId
	WHERE t.TripDate < GETDATE() 
	GROUP BY t.Title, t.TripDate
	ORDER BY t.TripDate DESC;
END
GO
