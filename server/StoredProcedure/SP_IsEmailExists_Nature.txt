CREATE PROCEDURE SP_IsEmailExists_Nature
	@Email NVARCHAR(100)
AS
BEGIN
	SET NOCOUNT ON;

	IF EXISTS (SELECT 1 FROM Users_Nature WHERE Email = @Email)
	BEGIN
		SELECT CAST(1 AS BIT) AS IsExists;
	END
	ELSE
	BEGIN
		SELECT CAST(0 AS BIT) AS IsExists;
	END
END
GO
