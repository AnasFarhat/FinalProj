CREATE PROCEDURE SP_GetUserByEmail_Nature 
	-- הוספת הפרמטר לאימייל
	@Email NVARCHAR(100)
AS
BEGIN
	-- מונע שליחת הודעות "X rows affected" שעלולות להפריע לקוד ב-C#
	SET NOCOUNT ON;

    -- השאילתה לשליפת המשתמש
	SELECT * FROM Users_Nature WHERE Email = @Email
END
