CREATE PROCEDURE SP_GetSentimentSummary_Nature
AS
BEGIN
	SET NOCOUNT ON;

	SELECT 
		ISNULL(SentimentStatus, 'Unknown') AS SentimentStatus, 
		COUNT(*) AS Count 
	FROM Feedbacks_Nature 
	GROUP BY SentimentStatus;
END
GO
