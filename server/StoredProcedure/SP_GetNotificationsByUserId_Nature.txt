CREATE PROCEDURE SP_GetNotificationsByUserId_Nature
	@UserId INT
AS
BEGIN
	SET NOCOUNT ON;

	SELECT Id, Title, Message, IsRead, CreatedAt 
	FROM Notifications_Nature 
	WHERE UserId = @UserId 
	ORDER BY CreatedAt DESC;
END
GO
