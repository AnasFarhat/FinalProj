CREATE PROCEDURE SP_ToggleLike_Nature
	@PostId INT,
	@UserId INT
AS
BEGIN
	SET NOCOUNT ON;

	IF EXISTS (SELECT 1 FROM PostLikes_Nature WHERE PostId = @PostId AND UserId = @UserId)
	BEGIN
		-- אם קיים - הסר
		DELETE FROM PostLikes_Nature WHERE PostId = @PostId AND UserId = @UserId;
		SELECT 'Like removed' AS Result;
	END
	ELSE
	BEGIN
		-- אם לא קיים - הוסף
		INSERT INTO PostLikes_Nature (PostId, UserId) VALUES (@PostId, @UserId);
		SELECT 'Like added' AS Result;
	END
END
GO
