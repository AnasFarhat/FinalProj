CREATE PROCEDURE SP_GetUserTrips_Nature
	@UserId INT
AS
BEGIN
	SET NOCOUNT ON;

	SELECT t.*, ut.Rating 
	FROM Trips_Nature t
	INNER JOIN UserTrips_Nature ut ON t.TripId = ut.TripId
	WHERE ut.UserId = @UserId;
END
GO
