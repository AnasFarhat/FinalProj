CREATE PROCEDURE SP_TogglePostVisibility_Nature
	@PostId INT
AS
BEGIN
	SET NOCOUNT ON;

	UPDATE Posts_Nature 
	SET IsHidden = CASE WHEN ISNULL(IsHidden, 0) = 1 THEN 0 ELSE 1 END 
	WHERE PostId = @PostId;


END
GO
